# tests/test_login.py
import pytest
from pages.login_page import LoginPage

@pytest.mark.parametrize("email,password,error", [
    ("user@test.com", "wrong", "Invalid credentials"),
    ("",              "123456", "Email is required"),
])
def test_login_negative(driver, email, password, error):
    page = LoginPage(driver).open()
    page.login(email, password)
    assert error in page.error_message()

      
# tests/test_users_api.py
import requests

BASE = "https://reqres.in/api"

def test_create_user_returns_201():
    payload = {"name": "Ali", "job": "QA Engineer"}
    r = requests.post(f"{BASE}/users", json=payload, timeout=10)

    assert r.status_code == 201
    body = r.json()
    assert body["name"] == payload["name"]
    assert "id" in body
